I added a ContextualTabGroup in code behind.
If I set the Visibility of the ContextualTabGroup to `Visibility.Visible` it won't show up. Only the dropdown button for the QuickAccessBar hides.

Here is the xaml and code behind file:

```
<fluent:RibbonWindow x:Class="FluentRibbonContextualTest.MainWindow"
        xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
        xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
        xmlns:d="http://schemas.microsoft.com/expression/blend/2008"
        xmlns:mc="http://schemas.openxmlformats.org/markup-compatibility/2006"
        xmlns:fluent="urn:fluent-ribbon"
        mc:Ignorable="d"
        Title="Contextual tab test" Height="350" Width="525">
    <fluent:RibbonWindow.Resources>
        <ResourceDictionary Source="pack://application:,,,/Fluent;Component/Themes/Office2013/generic.xaml"/>
    </fluent:RibbonWindow.Resources>
    <Grid>
        <fluent:Ribbon x:Name="RibbonControl" IsQuickAccessToolBarVisible="True"
                CanCustomizeQuickAccessToolBarItems="False" CanQuickAccessLocationChanging="False"
                CanCustomizeQuickAccessToolBar="False">
            <fluent:RibbonTabItem Header="Normal Tab"/>
        </fluent:Ribbon>
        <Button x:Name="BtnShow" Content="Show" HorizontalAlignment="Left" Margin="107,199,0,0" VerticalAlignment="Top" Width="75" Click="BtnShow_Click" />
        <Button x:Name="BtnHide" Content="Hide" HorizontalAlignment="Left" Margin="187,199,0,0" VerticalAlignment="Top" Width="75" Click="BtnHide_Click" />
    </Grid>
</fluent:RibbonWindow>
```

```
using Fluent;
using System.Windows;
using System.Windows.Media;

namespace FluentRibbonContextualTest
{
    public partial class MainWindow
    {
        public MainWindow()
        {
            InitializeComponent();
            AddNewContextualTab();
        }

        private void AddNewContextualTab()
        {
            RibbonContextualTabGroup tab = new RibbonContextualTabGroup()
            {
                Header = "Contextual Group",
                BorderBrush = Brushes.BlueViolet,
                Background = Brushes.BlueViolet
            };
            RibbonControl.ContextualGroups.Add(tab);

            RibbonTabItem item = new RibbonTabItem
            {
                Group = tab,
                Header = "Contextual Tab"
            };
        }

        private void BtnShow_Click(object sender, RoutedEventArgs e)
        {
            foreach (RibbonContextualTabGroup ribbonContextualTabGroup in RibbonControl.ContextualGroups)
            {
                ribbonContextualTabGroup.Visibility = Visibility.Visible;
            }
        }

        private void BtnHide_Click(object sender, RoutedEventArgs e)
        {
            foreach (RibbonContextualTabGroup ribbonContextualTabGroup in RibbonControl.ContextualGroups)
            {
                ribbonContextualTabGroup.Visibility = Visibility.Collapsed;
            }
        }
    }
}
```

This is a simple test application if you want to reproduce the issue.

I searched for a solution but i did not find something.. I am sorry if this question was already asked.
What I am doing wrong?
