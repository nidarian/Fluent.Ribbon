## TheMegaMatt commented on 2015-07-04T15:31:24Z

After a little bit of research i got to get this working almost fully (dont know how to hide the bottom border of the label) 

``` xaml
<ResourceDictionary xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
                    xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
                    xmlns:f="urn:fluent-ribbon">
    <Style TargetType="{x:Type TabItem}">
        <Setter Property="Template">
            <Setter.Value>
                <ControlTemplate TargetType="{x:Type TabItem}">
                    <Border x:Name="Border"
                            Margin="2,0"
                            BorderBrush="{DynamicResource RibbonTopBorderBrush}"
                            BorderThickness="0.75,0.75,0.75,0">
                        <ContentPresenter x:Name="ContentSite"
                                          Margin="20,5"
                                          HorizontalAlignment="Center"
                                          VerticalAlignment="Center"
                                          ContentSource="Header" />
                    </Border>
                    <ControlTemplate.Triggers>
                        <Trigger Property="IsSelected" Value="True">
                            <Setter Property="Foreground">
                                <Setter.Value>
                                    <SolidColorBrush Color="{StaticResource {x:Static f:MetroColors.ThemeColorKey}}" />
                                </Setter.Value>
                            </Setter>
                        </Trigger>
                        <Trigger Property="IsSelected" Value="False">
                            <Setter TargetName="Border" Property="Background" Value="White" />
                            <Setter TargetName="Border" Property="BorderThickness" Value="0,0,0,0" />
                        </Trigger>
                    </ControlTemplate.Triggers>
                </ControlTemplate>
            </Setter.Value>
        </Setter>
    </Style>
    <Style TargetType="{x:Type TabControl}">
        <Setter Property="BorderBrush" Value="{DynamicResource RibbonTopBorderBrush}" />
    </Style>
</ResourceDictionary>
```
