```

 public class SpinnerEx : Spinner
    {
        [SuppressMessage("Microsoft.Design", "CA1062", MessageId = "0")]
        protected override void OnMouseWheel(MouseWheelEventArgs e)
        {
            base.OnMouseWheel(e);
            Value += (e.Delta / 120) * Increment;
        }
    }

```
